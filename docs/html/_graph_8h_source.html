255template <class T>
256bool Vertex<T>::removeEdge(T in) {
257 bool removedEdge = false;
258 auto it = adj.begin();
259 while (it != adj.end()) {
260 Edge<T> *edge = *it;
261 Vertex<T> *dest = edge->getDest();
262 if (dest->getInfo() == in) {
263 it = adj.erase(it);
264 deleteEdge(edge);
265 removedEdge = true;
266 }
267 else {
268 it++;
269 }
270 }
271 return removedEdge;
272}

274template <class T>
275void Vertex<T>::removeOutgoingEdges() {
276 auto it = adj.begin();
277 while (it != adj.end()) {
278 Edge<T> *edge = *it;
279 it = adj.erase(it);
280 deleteEdge(edge);
281 }
282}

422template <class T>
423bool Graph<T>::removeVertex(const T &in) {
424 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
425 if ((*it)->getInfo() == in) {
426 auto v = *it;
427 v->removeOutgoingEdges();
428 for (auto u : vertexSet) {
429 u->removeEdge(v->getInfo());
430 }
431 vertexSet.erase(it);
432 delete v;
433 return true;
434 }
435 }
436 return false;
437}

458template <class T>
459bool Graph<T>::addBidirectionalEdge(const T &sourc, const T &dest, double w) {
460 auto v1 = findVertex(sourc);
461 auto v2 = findVertex(dest);
462 if (v1 == nullptr || v2 == nullptr)
463 return false;
464 auto e1 = v1->addEdge(v2, w);
465 auto e2 = v2->addEdge(v1, w);
466 e1->setReverse(e2);
467 e2->setReverse(e1);
468 return true;
469}
